== 3. MISFIRE: instants missed while Jod was down
==============================================================
-- fire_once must fire once for a long gap, not once per missed instant --
$ jod schedule rm contended
contended forgotten

$ jod schedule add missed-once say OK and stop --cron * * * * * --misfire fire_once
missed-once armed — next Aug 10 23:38 (in 15s)

-- backdate the next fire by three hours: 180 missed minutes --
$ q "SELECT name, misfire, next_fire_at_ms FROM schedules"
name | misfire | next_fire_at_ms
-----+---------+----------------
missed-once | fire_once | 1786394263062
(1 rows)

$ jod daemon --once

claimed 1 · started 1 · held 0 · failed 0

$ q "SELECT count(*) AS fires FROM schedule_fires"

(1 rows)

$ q "SELECT id, due_at_ms, outcome FROM schedule_fires ORDER BY id"
id | due_at_ms | outcome
---+-----------+--------
1 | 1786405020000 | ran
(1 rows)

-- and the next fire must be back in the future, not still three hours ago --
$ python3 - '/tmp/jod-e2e/dbreak/jod.db' <<'PY'
import sqlite3, sys, time, datetime
con = sqlite3.connect(f'file:{sys.argv[1]}?mode=ro', uri=True)
now = time.time()*1000
for name, nf in con.execute('SELECT name, next_fire_at_ms FROM schedules'):
    d = (nf - now)/1000
    print(f'{name}: next fire in {d:.0f}s ({"future" if d > 0 else "STILL IN THE PAST"})')
PY

== 5. kill the supervisor mid-run: does the run stay 'running' for ever?
==============================================================
$ jod schedule rm badcwd
badcwd forgotten

$ jod schedule rm missed-once
missed-once forgotten

$ jod schedule add killme count slowly to fifty, one number per line --cron * * * * *
killme armed — next Aug 10 23:38 (in 10s)

$ jod daemon --once
[jod/daemon] reloaded 4 prior run(s)
claimed 1 · started 1 · held 0 · failed 0

$ q "SELECT id, name, status, pid, pgid FROM runs ORDER BY created_at_ms DESC LIMIT 1"
id | name | status | pid | pgid
---+------+--------+-----+-----
59fb93d5-21e3-49aa-bd05-d27315c50a4d | killme | running | 1985960 | 1985960
(1 rows)

-- kill the supervisor's process group --
$ python3 - '/tmp/jod-e2e/dbreak/jod.db' <<'PY'
import sqlite3, sys, os, signal, time
con = sqlite3.connect(f'file:{sys.argv[1]}?mode=ro', uri=True)
row = con.execute("SELECT id, pgid FROM runs WHERE status='running' ORDER BY created_at_ms DESC LIMIT 1").fetchone()
if not row:
    print('no running run to kill'); sys.exit(0)
rid, pgid = row
print(f'killing pgid {pgid} for run {rid[:8]}')
os.killpg(pgid, signal.SIGKILL)
time.sleep(2)
PY
