==============================================================
== 6. forget destroys every version, and tombstones the deletion
==============================================================
-- before --
$ q "SELECT count(*) AS versions FROM facts WHERE subject='reljod' AND predicate='uses'"
versions
--------
2
(1 rows)

$ jod forget reljod uses
forgot 2 versions, permanently

-- after: zero rows of any version --
$ q "SELECT id, object, state FROM facts WHERE subject='reljod' AND predicate='uses'"
id | object | state
---+--------+------
(0 rows)

-- the FTS index must not keep a ghost --
$ jod recall jod on a vps
nothing remembered about that

-- a tombstone records that it happened, and how many versions died --
$ q "SELECT scope, subject, predicate, versions FROM tombstones"
scope | subject | predicate | versions
------+---------+-----------+---------
default | reljod | uses | 2
(1 rows)
