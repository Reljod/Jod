==============================================================

$ jod schedule add newyork morning brief --cron 0 2 * * * -z America/New_York

$ jod schedule ls
[32m●[0m [1mmanila              [0m [2m0 2 * * *       [0m Aug 11 18:00 (in 19h07m)
[32m●[0m [1mnewyork             [0m [2m0 2 * * *       [0m Aug 11 06:00 (in 7h07m)
[32m●[0m [1mnightly             [0m [2m0 2 * * *       [0m Aug 11 02:00 (in 3h07m)

-- the stored instant, rendered in UTC and in its own zone --
$ python3 - <<'PY'
import sqlite3, datetime, zoneinfo
con = sqlite3.connect('file:/tmp/jod-e2e/schedules/jod.db?mode=ro', uri=True)
for name, tz, ms in con.execute(
    'SELECT name, timezone, next_fire_at_ms FROM schedules ORDER BY name'):
    u = datetime.datetime.fromtimestamp(ms/1000, datetime.timezone.utc)
    l = u.astimezone(zoneinfo.ZoneInfo(tz))
    print(f'{name:10} tz={tz:18} utc={u:%Y-%m-%d %H:%M} local={l:%Y-%m-%d %H:%M %Z}')
PY
manila     tz=Asia/Manila        utc=2026-08-11 18:00 local=2026-08-12 02:00 PST
newyork    tz=America/New_York   utc=2026-08-11 06:00 local=2026-08-11 02:00 EDT
nightly    tz=UTC                utc=2026-08-11 02:00 local=2026-08-11 02:00 UTC
